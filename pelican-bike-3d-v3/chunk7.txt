nst ang = Math.acos(cosKnee);
      const shoulder = base + ang * leg.side;
      const kneeX = root.x + Math.cos(shoulder)*leg.upperLen;
      const kneeY = root.y + Math.sin(shoulder)*leg.upperLen;
      const lower = Math.atan2(tar.y-kneeY, tar.x-kneeX);

      leg.upperG.position.copy(leg.root.position); leg.upperG.position.set(0,0,0);
      leg.upperG.rotation.z = shoulder - Math.PI/2;
      leg.lowerG.position.set(kneeX-root.x, kneeY-root.y, 0);
      leg.lowerG.rotation.z = lower - Math.PI/2;
      leg.footG.position.set(tar.x-root.x, tar.y-root.y, 0); leg.footG.rotation.z = .1 + Math.sin(performance.now()*.01)*.02;
      leg.knee.set(kneeX,kneeY,leg.side*.02);
    }

    const clock = new THREE.Clock(); let elapsed=0; let fpsAcc=0,fpsFrames=0;
    function animate(){
      requestAnimationFrame(animate);
      let dt = Math.min(clock.getDelta(), .05); if(paused) dt=0; elapsed += dt;
      const t = elapsed; fpsAcc += dt; fpsFrames++; if(fpsAcc>.4){ fpsText.textContent = Math.round(fpsFrames/fpsAcc); fpsAcc=0; fpsFrames=0; }
      boost = Math.max(0, boost - dt*1.25);
      const rideSpeed = speed * (1 + boost*.58);
      const k = rideSpeed/20;

      // bike and rider motion
      const rot = dt * (rideSpeed/3.2);
      for(const w of wheels) w.rotation.z -= rot;
      pedalGroup.rotation.z -= rot*1.55;
      chainring.rotation.z = pedalGroup.rotation.z; rearCog.rotation.z = -pedalGroup.rotation.z*.34;
      bike.position.y = .025 + Math.sin(t*7*k)*.018*(rideSpeed>0); bike.rotation.z = Math.sin(t*2.3)*.008 + boost*.01;
      blobShadow.material.opacity = .32 + Math.abs(bike.position.y)*3.8; blobShadow.scale.setScalar(1 + Math.abs(bike.position.y)*1.6);
      body.scale.y = 1.3 + Math.sin(t*5.7*k)*.018;
      headPivot.rotation.z = Math.sin(t*2.2)*.05 + boost*.02; headPivot.rotation.y = Math.sin(t*1.2)*.035;
      wingL.rotation.x = .12 + Math.sin(t*2.45)*.1 + wind*.16; wingR.rotation.x = -.12 - Math.sin(t*2.45)*.1 - wind*.16;
      wingL.rotation.z = .02 + Math.sin(t*1.6)*.05; wingR.rotation.z = -.02 - Math.sin(t*1.6)*.05;
      scarfTail1.rotation.z = .18 + Math.sin(t*(8+wind*4))* (.16 + wind*.18) + boost*.06;
      scarfTail2.rotation.z = .28 + Math.sin(t*(10+wind*6)+1.5) * (.14 + wind*.16) + boost*.09;
      tailPivot.rotation.z = -.08 + Math.sin(t*5)*.08;
      for(const f of bodyFeathers){ f.rotation.y = Math.sin(t*3 + f.userData.phase)*(.06 + wind*.08); f.position.y = f.userData.baseY + Math.sin(t*2.5+f.userData.phase)*.01; }
      headPivot.children.forEach(ch => { if(ch.userData.crest!==undefined) ch.rotation.z = -1.15 + ch.userData.crest*.12 + Math.sin(t*6 + ch.userData.crest)*(.08 + wind*.12); });

      // IK pedal targets
      for(let i=0;i<legRig.length;i++){
        const phase = i===0 ? 0 : Math.PI;
        const ang = -pedalGroup.rotation.z + phase;
        const p = new THREE.Vector3(.1 + Math.cos(ang)*.65, 1.12 + Math.sin(ang)*.02, i===0?.25:-.25);
        solveLeg(legRig[i], p);
      }

      // chain motion
      for(let i=0;i<chainLinks.length;i++){
        const u = i/chainLinks.length - pedalGroup.rotation.z/(Math.PI*2)*.12;
        const p1 = chainPoint(u), p2 = chainPoint(u+.01);
        chainLinks[i].position.copy(p1); chainLinks[i].rotation.z = Math.atan2(p2.y-p1.y, p2.x-p1.x);
      }

      // move environment
      const dx = dt*rideSpeed*.47;
      for(const m of laneMarks){ m.position.x -= dx; if(m.position.x<-72) m.position.x+=156; }
      for(const m of edgeMarks){ m.position.x -= dx; if(m.position.x<-78) m.position.x+=148; }
      for(const o of worldMoving){ o.position.x -= dx*.75; if(o.position.x<-84) o.position.x += 168; }
      clouds.forEach((c,i)=>{ c.position.x -= dt*(.2 + i*.01 + rideSpeed*.006); if(c.position.x<-58) c.position.x=62; });
      flocks.forEach((f,idx)=>{ f.position.x -= dt*(.8 + idx*.25); if(f.position.x<-32) f.position.x=36; f.children.forEach((b,j)=>{ b.rotation.z = Math.sin(t*8 + b.userData.phase + j)*b.userData.amp; }); });

      // dust
      const a = dustGeo.attributes.position.array;
      for(let i=0;i<dustCount;i++){
        a[i*3] -= dt*(2 + rideSpeed*.22); a[i*3+1] += dt*.08*Math.sin(i+t*4); if(a[i*3]<-13){ a[i*3]=-2-Math.random()*2; a[i*3+1]=.1+Math.random()*.45; a[i*3+2]=(Math.random()-.5)*3.4; }
      }
      dustGeo.attributes.position.needsUpdate = true;

      // speed lines
      const sl = speedLinesGeo.attributes.position.array; const lineOpacity = clamp((rideSpeed-18)/30,0,1) * (.36 + boost*.18); speedLines.material.opacity = lineOpacity;
      for(let i=0;i<lineCount;i++){
        const x = 2 + (i/lineCount)*16 + ((t*rideSpeed*.9 + i*1.7)%16);
        const y = 1.4 + Math.sin(i*2.1+t*3)*2.4;
        const z = -1.8 - (i%7)*.5;
        sl[i*6+0]=x; sl[i*6+1]=y; sl[i*6+2]=z;
        sl[i*6+3]=x-1.2 - boost*.8; sl[i*6+4]=y; sl[i*6+5]=z;
      }
      speedLinesGeo.attributes.position.needsUpdate = true;

      // sea time
      seaMat.uniforms.time.value = t;

      distance += dt*rideSpeed/3600; distanceText.textContent = distance.toFixed(2)+' km';
      speedText.textContent = rideSpeed.toFixed(1);
      updateLighting(); updateCamera(dt,t);
      if(bloomEnabled) composer.render(); else renderer.render(scene,camera);
    }

    addEventListener('resize', ()=>{ camera.aspect=innerWidth/innerHeight; camera.updateProjectionMatrix(); renderer.setSize(innerWidth,innerHeight); composer.setSize(innerWidth,innerHeight); });
    controls.enabled=false; updateLighting(); animate();
    setTimeout(()=>$('#loader').classList.add('hide'),460);
