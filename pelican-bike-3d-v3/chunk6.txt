    function updateCamera(dt,t){
      let mode = camMode;
      let shotKey = 'MAN';
      if(camMode==='director' && !directorLocked){ const shot = activeShot(t*.34 + boost*.7); mode = shot.mode; shotKey = shot.key; }
      shotText.textContent = shotKey;
      if(mode==='orbit'){ controls.update(); return; }
      if(mode==='follow'){ desiredPos.set(-8.15,4.75,9.6); desiredTarget.set(.1,2.62,0); }

      else if(mode==='drone'){ desiredPos.set(-1.1,10.8,4.2); desiredTarget.set(0,1.4,0); }
      if(camMode==='director' && !directorLocked && mode!=='orbit' && shotKey==='C3'){ cinemaAngle += dt*.75; desiredPos.x += Math.cos(cinemaAngle)*.85; desiredPos.z += Math.sin(cinemaAngle)*.95; }

      camera.updateProjectionMatrix();
      camera.position.lerp(desiredPos,1-Math.exp(-dt*2.8)); controls.target.lerp(desiredTarget,1-Math.exp(-dt*3.1)); camera.lookAt(controls.target);
    }
